# 八、复现完整流水线（约 10 分钟）

```bash
# 1. 语料清洗
python3 02_代码与本体/kg-service/clean_corpus.py \
    --input 04_参考资料/原文_2024共识_全文.txt \
    --out /tmp/2024共识_clean.txt

# 2. 本体构建 + 三元组抽取
python3 02_代码与本体/kg-service/ontology_builder.py \
    --input /tmp/2024共识_clean.txt \
    --out-ontology /tmp/seed.json --out-triples /tmp/triples.jsonl

# 3. 覆盖率审计
python3 02_代码与本体/kg-service/ontology_coverage_audit.py \
    --corpus /tmp/2024共识_clean.txt --builder 02_代码与本体/kg-service/ontology_builder.py

# 4. 接线审计（找出"定义了但抽取不到"的类目）
python3 02_代码与本体/kg-service/ontology_wiring_audit.py \
    --template 02_代码与本体/kg-service/workflow_template.py \
    --spec medical --corpus /tmp/2024共识_clean.txt

# 5. 多行业迁移验证
python3 02_代码与本体/kg-service/workflow_template.py --validate \
    --corpus-dir 04_参考资料/ --out /tmp/migration.json

# 6. 引擎层评测
python3 02_代码与本体/eval/eval_runner.py \
    --evalset 03_验证与数据/自测评测集_58条.json

# 7. 全包一致性扫描
python3 02_代码与本体/tools/consistency_scan.py --pkg 提交包_初赛
```
